#!/usr/bin/env bash

set -euo pipefail

REPO="${GS_REPO:-CorniiDog/gamescope-nvidia}"
BRANCH="${GS_BRANCH:-master}"

TMP_DIR="$(mktemp -d)"

cleanup()
{
    rm -rf "$TMP_DIR"
}

trap cleanup EXIT

need()
{
    command -v "$1" >/dev/null 2>&1 || {
        echo "[gamescope-nvidia] Missing command: $1" >&2
        exit 1
    }
}

need curl
need tar

echo "[gamescope-nvidia] Downloading ${REPO}:${BRANCH}..."

ARCHIVE="${TMP_DIR}/source.tar.gz"

curl \
    -fL \
    --retry 3 \
    "https://github.com/${REPO}/archive/refs/heads/${BRANCH}.tar.gz" \
    -o "$ARCHIVE"

tar \
    -xzf "$ARCHIVE" \
    -C "$TMP_DIR"

PROJECT_DIR="$(
    find "$TMP_DIR" \
        -mindepth 1 \
        -maxdepth 1 \
        -type d \
        -name 'gamescope-nvidia-*' \
        -print \
        -quit
)"

[[ -n "$PROJECT_DIR" ]] || {
    echo "[gamescope-nvidia] Could not locate extracted source tree." >&2
    exit 1
}

chmod +x \
    "${PROJECT_DIR}/bootstrap/setup.sh" \
    "${PROJECT_DIR}/bootstrap/build.sh" \
    "${PROJECT_DIR}/bootstrap/install.sh" \
    "${PROJECT_DIR}/bootstrap/uninstall.sh"

exec \
    "${PROJECT_DIR}/bootstrap/setup.sh" \
    "$@"
